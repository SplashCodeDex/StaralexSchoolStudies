\documentclass[tikz,border=6pt]{standalone}
\usepackage{mathdiagrams}
% surd_area_simplification_18 — visual proof that sqrt(18) = 3*sqrt(2)
\begin{document}
\begin{tikzpicture}[md]
  % ---- Left square: Area 18, side sqrt(18) ----
  \fill[mdFill] (0, 0) rectangle (3.6, 3.6);
  \draw[md thick] (0, 0) rectangle (3.6, 3.6);
  \node[md label bold] at (1.8, 1.8) {Area $= 18$};
  \draw[md brace, decorate, decoration={brace, mirror, amplitude=5pt, raise=3pt}]
    (0, 0) -- (3.6, 0) node[midway, below=9pt, md label bold] {$\sqrt{18}$};
  \draw[md brace, decorate, decoration={brace, amplitude=5pt, raise=3pt}]
    (0, 0) -- (0, 3.6) node[midway, left=9pt, md label bold] {$\sqrt{18}$};

  % Equals sign between panels
  \node[md label big] at (4.4, 1.8) {$=$};

  % ---- Right square: 3x3 grid of area-2 tiles (side sqrt(2)) ----
  \fill[mdAccentSoft, fill opacity=0.45] (5.2, 0) rectangle (8.8, 3.6);
  \foreach \i in {0, 1, 2} {
    \foreach \j in {0, 1, 2} {
      \pgfmathsetmacro{\x}{5.2 + \i*1.2}
      \pgfmathsetmacro{\y}{\j*1.2}
      \draw[md thin, draw=mdAccent] (\x, \y) rectangle (\x+1.2, \y+1.2);
      \node[md label muted] at (\x+0.6, \y+0.6) {$2$};
    }
  }
  \draw[md thick, draw=mdAccent] (5.2, 0) rectangle (8.8, 3.6);

  % 3 individual sqrt(2) braces along the bottom
  \foreach \i in {0, 1, 2} {
    \pgfmathsetmacro{\x}{5.2 + \i*1.2}
    \draw[md brace, color=mdAccent, decorate, decoration={brace, mirror, amplitude=4pt, raise=3pt}]
      (\x+0.03, 0) -- (\x+1.17, 0) node[midway, below=8pt, md label accent] {$\sqrt{2}$};
  }
  % Total width label on top
  \draw[md brace, color=mdAccent, decorate, decoration={brace, amplitude=5pt, raise=3pt}]
    (5.2, 3.6) -- (8.8, 3.6) node[midway, above=9pt, md label accent] {$3\sqrt{2}$};
\end{tikzpicture}
\end{document}
